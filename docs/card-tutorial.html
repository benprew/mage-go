Always set Outcome. The AI uses it for targeting.

Swords to Plowshares

Swords to Plowshares {W} Instant Exile target creature. Its controller gains life equal to its power. 
mage.NewInstant("Swords to Plowshares", "{W}",
  mage.NewTargetedSpell(
    mage.TargetCreature(),
    mage.FuncEffect(
      "exile; controller gains life = power",
      mage.EffectProperties{
          Outcome: mage.OutcomeDetriment},
      func(g mage.GameMutator,
           srcID, ctrl uuid.UUID,
           targets []uuid.UUID) error {
        if len(targets) == 0 { return nil }
        perm := g.FindPermanent(targets[0])
        if perm == nil { return nil }
        power := perm.CurrentPower(g)
        g.ExilePermanent(perm)
        p := g.GetPlayer(perm.Controller)
        if p != nil && power > 0 {
            g.PlayerGainLife(p, power)
        }
        return nil
      },
    ),
  ),
)

ReplacementEffect Interface

type ReplacementEffect interface {
    Matches(Action, GameReader) bool
    Replace(Action, GameMutator) Action
    SourceID() uuid.UUID
    IsActive(GameReader) bool
}

Replace returns:

• Modified action (reduce damage)

• Different type (redirect player → creature)

• nil (fully prevent)

type halveDamageReplacement struct {
    replacementBase
    playerID uuid.UUID
}
func (r *halveDamageReplacement) Matches(a Action, _ GameReader) bool {
    act, ok := a.(*DamageToPlayerAction)
    return ok && act.PlayerID() == r.playerID
}
func (r *halveDamageReplacement) Replace(a Action, _ GameMutator) Action {
    act := a.(*DamageToPlayerAction)
    return act.WithAmount(act.Amount() / 2)
}
func (r *halveDamageReplacement) IsActive(_ GameReader) bool { return true }

Five Testing Patterns

1. Positive AND negative

t.Run("triggers_on_own", ...)

2. Source removal

t.Run("removing_king_removes_boost", ...)

3. Stack interaction

g.CastSpell(..., "Lightning Bolt", ...)
g.CastInResponseTo(PA, "Counterspell")

4. Across turns

// 3 upkeeps of Juzam (1 dmg each)
g.StopAt(5, core.PrecombatMain)
g.AssertLife(PA, 17)

5. Combat math

// First strike kills before normal
g.Attack(1, PA, "White Knight")
g.Block(1, PB, "Bears", "Knight")
// Knight survives, Bears die
